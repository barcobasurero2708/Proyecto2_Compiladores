fun int evaluar_notas(int n1, int n2, int n3, int n4)
    var int aprobadas;
    aprobadas = count(60 <= n1, 60 <= n2, 60 <= n3, 60 <= n4);
    return aprobadas
endfun

fun bool es_sobresaliente(int n1, int n2, int n3, int n4)
    var int altas;
    altas = count(85 <= n1, 85 <= n2, 85 <= n3, 85 <= n4);
    return 2 <= altas
endfun

fun int main()
    var int aprob, nota_min, nota_max;
    var bool honor;
    aprob = evaluar_notas(85, 42, 90, 58);
    print(aprob);

    nota_min = min(85, 42, 90, 58);

endfun
