#include <iostream>

using namespace std;

int main() {
    int a = 3, b = 5, i = 0, suma = 0;
    cout << (a < b) << endl;
    cout << (a > b) << endl;
    cout << (a >= 3) << endl;
    cout << (a == b) << endl;
    cout << (a != b) << endl;
    cout << (a < b == b > a) << endl;
    while (i < 5) {
        suma = suma + i;
        i = i + 1;
    }
    cout << suma << endl;
    do {
        i = i - 1;
    } while (i > 2);
    cout << i << endl;
    return 0;
}
