#include <iostream>
using namespace std;

// switch: caída entre casos, default en medio, break y continue
int puntos(char letra) {
    switch (letra) {
        case 'a':
        case 'A':
            return 10;
        case 'b':
            return 5;
        default:
            return 0;
    }
}

int main() {
    cout << puntos('A') << " " << puntos('b') << " " << puntos('z') << endl;
    for (int i = 0; i < 6; i++) {
        switch (i) {
            case 1:
                continue;
            case 2:
                cout << "dos ";
            case 3:
                cout << "tres ";
                break;
            default:
                cout << "otro ";
            case 2 + 3:
                cout << "cinco ";
        }
        cout << "| " << i << endl;
    }
    return 0;
}
