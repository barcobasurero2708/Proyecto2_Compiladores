fun int find_min_recursive(int n, int current_min)
    var int val;
    if n <= 0 then
        return current_min
    endif;
    val = n * n - 6 * n + 5;
    return find_min_recursive(n - 1, min(current_min, val))
endfun

fun int find_max_recursive(int n, int current_max)
    var int val;
    if n <= 0 then
        return current_max
    endif;
    val = n * n - 6 * n + 5;
    return find_max_recursive(n - 1, max(current_max, val))
endfun

fun int main()
    var int m1, m2;
    m1 = find_min_recursive(6, 1000);
    m2 = find_max_recursive(6, 0 - 1000);
    print(m1);
    print(m2);
    print(m2 - m1);
    return 0
endfun
