#include <iostream>

using namespace std;

int main() {
    int n = 0, pares = 0;
    do {
        n = n + 1;
        if (n - n / 2 * 2 <= 0) {
            pares = pares + 1;
        }
    } while (n <= 9);
    cout << n << endl;
    cout << pares << endl;
    do {
        n = n - 1;
    } while (100 <= n);
    cout << n << endl;
    return 0;
}
