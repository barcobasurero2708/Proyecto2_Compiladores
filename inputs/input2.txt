#include <iostream>

using namespace std;

int factorial(int n) {
    if (n <= 1) {
        return 1;
    } else {
        return n * factorial(n - 1);
    }
}

int fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}

int main() {
    cout << factorial(5) << endl;
    cout << factorial(10) << endl;
    cout << fib(10) << endl;
    return 0;
}
