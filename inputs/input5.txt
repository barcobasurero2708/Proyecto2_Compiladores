#include <iostream>

using namespace std;

int clasificar(int nota) {
    if (nota <= 59) {
        return 0;
    } else if (nota <= 79) {
        return 1;
    } else if (nota <= 89) {
        return 2;
    } else {
        return 3;
    }
}

bool esPositivo(int x) {
    return 1 <= x;
}

int main() {
    cout << clasificar(40) << endl;
    cout << clasificar(70) << endl;
    cout << clasificar(85) << endl;
    cout << clasificar(95) << endl;
    bool p = esPositivo(-5);
    cout << (p ? 1 : 0) << endl;
    cout << (esPositivo(8) ? 111 : 222) << endl;
    return 0;
}
