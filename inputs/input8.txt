#include <iostream>

using namespace std;

int main() {
    int i = 1, j, acum = 0;
    while (i <= 3) {
        j = 1;
        while (j <= 3) {
            if (j <= 1) {
                j = j + 1;
            } else {
                acum = acum + i * j;
                j = j + 1;
            }
        }
        i = i + 1;
    }
    cout << acum << endl;
    cout << max(3, 9, 4) << endl;
    cout << min(3, 9, 4) << endl;
    cout << count(1 <= 2, 5 <= 2, true) << endl;
    return 0;
}
